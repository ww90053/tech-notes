6.Harbor建置

在安裝harbor前須先安裝docker及docker-compose

# 更新系統套件
sudo apt update
# 安裝docker所需依賴套件
sudo apt install -y apt-transport-https ca-certificates curl software-properties-common
# 匯入docker官方gpg金鑰
curl -fsSL https://download.docker.com/linux/ubuntu/gpg | sudo gpg --dearmor -o /usr/share/keyrings/docker-archive-keyring.gpg
# 新增docker repository
echo "deb [arch=$(dpkg --print-architecture) signed-by=/usr/share/keyrings/docker-archive-keyring.gpg] https://download.docker.com/linux/ubuntu $(lsb_release -cs) stable" | sudo tee /etc/apt/sources.list.d/docker.list > /dev/null
# 安裝docker
sudo apt update
sudo apt install -y docker-ce docker-ce-cli containerd.io
# 安裝docker-compose
sudo apt install -y docker-compose
# 驗證安裝
docker --version
docker-compose --version
# 重啟docker
sudo systemctl restart docker

以下指令執行前請先切換至root角色

mdkir -p /root/harbor
cd /root/harbor
# 可以從github找到最新版本 https://github.com/goharbor/harbor/releases
curl -L -o harbor-online-installer-v2.12.2.tgz https://github.com/goharbor/harbor/releases/download/v2.12.2/harbor-online-installer-v2.12.2.tgz
# 解壓縮
tar -xzvf harbor-online-installer-v2.12.2.tgz
# 壓縮檔內容
# harbor/prepare
# harbor/LICENSE
# harbor/install.sh
# harbor/common.sh
# harbor/harbor.yml.tmpl

生成自簽憑證(非必要)，第二台請更換成k8shb2

mkdir -p /root/harbor/ca
cd /root/harbor/ca
# 生成 CA 私鑰
openssl genrsa -out ca.key 4096
# 生成 CA 證書
openssl req -x509 -new -nodes -sha512 -days 3650 \
-subj "/C=TW/ST=Taiwan/L=Taipei/O=Iscom/OU=IT/CN=k8shb.hccg.gov.tw" \
-key ca.key \
-out ca.crt
# 生成服務器私鑰
openssl genrsa -out harbor.key 4096cd
# 生成服務器證書請求
openssl req -sha512 -new \
-subj "/C=TW/ST=Taiwan/L=Taipei/O=Iscom/OU=IT/CN=k8shb.hccg.gov.tw" \
-key harbor.key \
-out harbor.csr
# 生成x509 v3 extension檔案
cat > v3.ext <<-EOF
authorityKeyIdentifier=keyid,issuer
basicConstraints=CA:FALSE
keyUsage = digitalSignature, nonRepudiation, keyEncipherment, dataEncipherment
extendedKeyUsage = serverAuth
subjectAltName = @alt_names

[alt_names]
DNS.1=k8shb.hccg.gov.tw
DNS.2=harbor.iscom.com
DNS.3=harbor
EOF
# 生成服務器證書
openssl x509 -sha512 -req -days 3650 \
-extfile v3.ext \
-CA ca.crt -CAkey ca.key -CAcreateserial \
-in harbor.csr \
-out harbor.crt

安裝Harbor

第二台請修改hostname換成k8shb2

# 複製模板
cp harbor.yml.tmpl harbor.yml
# 調整參數
nano harbor.yml
# harbor.yml
hostname: k8shb.hccg.gov.tw 
http:
  port: 80
https:
  port: 443
  certificate: /root/harbor/ca/harbor.crt
  private_key: /root/harbor/ca/harbor.key
harbor_admin_password: iscom@70598966
# 儲存後執行安裝
sudo ./install.sh
